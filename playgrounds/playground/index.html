                <script>
                    props({ count: 0 })

                    function inc() {
                        this.state.count++
                    }

                    function dec() {
                        this.state.count = Math.max(0, this.state.count - 1)
                    }
                </script>

                <template>
                    <div>
                        <p>Count: <strong>{{ count }}</strong></p>
                        <button type="button" @click(this.dec())">−</button>
                        <button type="button" @click(this.inc())">+</button>
                    </div>
                </template>
            

            
                <script>
                    props({
                        names: ['Alice', 'Bob', 'Carlo'],
                        selected: null,
                    })

                    function greet(e) {
                        this.state.selected = e.currentTarget.getAttribute('data-person')
                    }
                </script>

                <style scoped>
                    .row {
                        display: flex;
                        flex-wrap: wrap;
                        gap: 0.4rem;
                        align-items: center;
                    }

                    .msg {
                        margin: 0.5rem 0 0;
                        font-weight: 600;

                    }
                </style>

                <template>
                    <div style-scoped class="row">
                        @foreach(names as person)
                            <button type="button" data-person="{{ person }}" @click(this.greet(event))>{{ person }}</button>
                        @endforeach
                        @if(selected)
                            <p class="msg">Hello, {{ selected }}!</p>
                        @endif
                    </div>
                </template>
            

            
                <script>
                    props({ items: ['Learn Jslade', 'Edit this list'], draft: '' })

                    function onDraft(e) {
                        this.draft = e.target.value
                    }

                    function add() {
                        var text = (this.draft || '').trim()
                        if (!text) return
                        this.state.items = this.state.items.concat(text)
                        this.draft = ''
                    }

                    function remove(e) {
                        var i = Number(e.currentTarget.getAttribute('data-i'))
                        this.state.items = this.state.items.filter(function (_, idx) { return idx !== i })
                    }
                </script>

                <template>
                    <div>
                        <ul>
                            @foreach(items as item)
                                <li>
                                    {{ item }}
                                    <button type="button" data-i="{{ $loop.index }}" @click(this.remove(event))">×</button>
                                </li>
                            @endforeach
                        </ul>
                        <input type="text" value="{{ draft }}" @input(this.onDraft(event)) placeholder="New item…">
                        <button type="button" @click(this.add())">Add</button>
                    </div>
                </template>
